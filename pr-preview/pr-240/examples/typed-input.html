Example: typed input ​

Four handlers extend TypedHandler. The DTO on handleDto() is the request contract: #[Query], #[Path], #[Header], #[Body] and #[File]. The same order endpoint accepts JSON and a URL-encoded form. See Typed input.

Runnable project

Every file below is imported from examples/typed-input. Clone the repository, cd examples/typed-input, run composer install and then ./e2e.sh, which starts the PHP built-in server and checks every response shown on this page.

Key files

  • config/common.neon
  • app/Api/ListUsersHandler.php
  • app/Api/CreateOrderHandler.php
  • app/Api/UserDetailHandler.php
  • app/Api/AvatarUploadHandler.php
  • e2e.sh
View source on GitHub

A query DTO ​

php
<?php

declare(strict_types=1);

namespace App\Api\Input;

use Tomaj\NetteApi\Input\Query;
use Tomaj\NetteApi\Validation\Range;

final readonly class ListUsersInput
{
    public function __construct(
        #[Query(description: 'Page number')]
        #[Range(min: 1)]
        public int $page = 1,
        #[Query]
        public ?UserStatus $status = null,
    ) {
    }
}
php
<?php

declare(strict_types=1);

namespace App\Api;

use App\Api\Input\ListUsersInput;
use Tomaj\NetteApi\Attribute\Get;
use Tomaj\NetteApi\Handlers\TypedHandler;
use Tomaj\NetteApi\Response\JsonApiResponse;
use Tomaj\NetteApi\Response\ResponseInterface;

#[Get(1, 'users', 'list')]
final class ListUsersHandler extends TypedHandler
{
    public function handleDto(ListUsersInput $input): ResponseInterface
    {
        return new JsonApiResponse(200, [
            'page' => $input->page,
            'status' => $input->status?->value,
        ]);
    }
}

A file upload ​

php
<?php

declare(strict_types=1);

namespace App\Api\Input;

use Tomaj\NetteApi\Input\Body;
use Tomaj\NetteApi\Input\File;
use Tomaj\NetteApi\Input\FileUpload;
use Tomaj\NetteApi\Validation\Length;

final readonly class AvatarInput
{
    public function __construct(
        #[Body]
        #[Length(min: 1, max: 80)]
        public string $displayName,
        #[File(description: 'PNG or JPEG image')]
        public FileUpload $avatar,
    ) {
    }
}
php
<?php

declare(strict_types=1);

namespace App\Api;

use App\Api\Input\AvatarInput;
use Tomaj\NetteApi\Attribute\Post;
use Tomaj\NetteApi\Handlers\TypedHandler;
use Tomaj\NetteApi\Response\JsonApiResponse;
use Tomaj\NetteApi\Response\ResponseInterface;

#[Post(1, 'avatar')]
final class AvatarUploadHandler extends TypedHandler
{
    public function handleDto(AvatarInput $input): ResponseInterface
    {
        $avatar = $input->avatar;
        if ($avatar->error !== UPLOAD_ERR_OK) {
            return new JsonApiResponse(400, ['status' => 'error', 'message' => 'upload failed']);
        }

        return new JsonApiResponse(201, [
            'name' => $input->displayName,
            'file' => $avatar->originalName,
            'size' => $avatar->size,
        ]);
    }
}

Registration ​

The path endpoint is #[Get(1, 'users', '{id}/detail')]. The route is the usual one, with an optional third segment:

php
<?php

declare(strict_types=1);

namespace App;

use Nette\Application\Routers\RouteList;

final class RouterFactory
{
    public static function create(): RouteList
    {
        $router = new RouteList();
        // {id} in users/{id}/detail is one more segment, captured by <params>
        $router->addRoute('/api/v<version>/<package>[/<apiAction>][/<params>]', 'Api:Api:default');
        return $router;
    }
}
neon
extensions:
	api: Tomaj\NetteApi\DI\NetteApiExtension

services:
	router: App\RouterFactory::create
	apiLink: Tomaj\NetteApi\Link\ApiLink

api:
	outputConfigurator: Tomaj\NetteApi\Output\Configurator\EnvConfigurator('APP_ENV', 'production')

	discovery:
		paths: [%appDir%/Api]
		namespace: App\Api

	endpoints:
		- {method: GET, version: 1, package: openapi, handler: Tomaj\NetteApi\Handlers\OpenApiHandler, authorization: false}
